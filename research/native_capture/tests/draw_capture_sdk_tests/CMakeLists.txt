cmake_minimum_required(VERSION 3.20)
project(draw_capture_sdk_tests LANGUAGES CXX)
if(NOT DEFINED REXGLUE_SDK_ROOT OR
   NOT EXISTS "${REXGLUE_SDK_ROOT}/include/rex/graphics/vulkan/draw_identity_capture.h")
  message(FATAL_ERROR "REXGLUE_SDK_ROOT must point to a patch-applied isolated SDK worktree")
endif()
find_package(Threads REQUIRED)
foreach(test_name IN ITEMS draw_identity_filter_test draw_identity_selector_test)
  add_executable(${test_name} "../${test_name}.cpp")
  target_include_directories(${test_name} PRIVATE "${REXGLUE_SDK_ROOT}/include")
  target_compile_features(${test_name} PRIVATE cxx_std_23)
  target_compile_options(${test_name} PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(${test_name} PRIVATE Threads::Threads)
  enable_testing()
  add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
