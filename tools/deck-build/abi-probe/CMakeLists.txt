cmake_minimum_required(VERSION 3.25)
project(fable_deck_abi_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
add_library(deck_abi_probe SHARED abi_library.cpp)
target_compile_options(deck_abi_probe PRIVATE -march=x86-64-v2 -O2 -fvisibility=hidden)
target_compile_definitions(deck_abi_probe PRIVATE ABI_LIBRARY_BUILD=1)
set_target_properties(deck_abi_probe PROPERTIES OUTPUT_NAME deck_abi_probe)
add_executable(deck_abi_probe_test abi_main.cpp)
target_compile_options(deck_abi_probe_test PRIVATE -march=x86-64-v2 -O2)
target_link_libraries(deck_abi_probe_test PRIVATE deck_abi_probe)
set_target_properties(deck_abi_probe_test PROPERTIES BUILD_RPATH "$ORIGIN")
enable_testing()
add_test(NAME deck_abi_probe_runtime COMMAND deck_abi_probe_test)
